2. Bash essentials

#!/usr/bin/env bash
set -euo pipefail

ps aux | grep nginx | awk '{print $2}'
df -h | tee /var/log/disk-report.txt
grep -E 'ERROR|FATAL' app.log | sed 's/^/ALERT /'

# backup kata
tar -czf "/backup/home-$(date +%F).tar.gz" /home/app
# cron: 0 2 * * * /usr/local/bin/backup-home.sh

  • Never commit secrets; use env files outside git

4. Groovy awareness (Jenkins)

pipeline {
  agent any
  stages {
    stage('Test') { steps { sh 'pytest -q' } }
    stage('Build') { steps { sh 'docker build -t $APP:$GIT_COMMIT .' } }
    stage('Deploy') {
      steps {
        // ansiblePlaybook / azureCLI / kubectl — via plugins or sh
        sh 'ansible-playbook -i invent deploy.yml'
      }
    }
  }
  post { failure { echo 'Notify on-call' } }
}
  • Declarative stages + agents; sh reuses your Bash
  • Wire Git checkout → test → Ansible/Terraform/Docker
  • Credentials IDs only — never hardcode
